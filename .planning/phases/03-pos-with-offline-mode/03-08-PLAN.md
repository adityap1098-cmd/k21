For simplicity in Phase 3: if no serverId is stored in the conflict row, "Batalkan" simply deletes the IndexedDB record and writes a local void audit (no server call needed since no server record was created).

    Since this file will use hooks (useEffect for sync listener, useState for modal state), make the interactive part a Client Component. The page.tsx itself can be a thin Server Component that imports a 'PosPageClient' client component. Or: use 'use client' at the top of page.tsx directly (App Router allows this).

    Full POS page structure:
    ```typescript
    'use client'

    import { useEffect, useState } from 'react'
    import { useShiftStore } from '@/lib/store/shift.store'
    import { startSyncListener } from '@/lib/sync-manager'
    import { ProductPanel } from '@/components/pos/ProductPanel'
    import { CartPanel } from '@/components/pos/CartPanel'
    import { PaymentModal } from '@/components/pos/PaymentModal'
    import { ReceiptModal } from '@/components/pos/ReceiptModal'
    import { ShiftDrawer } from '@/components/pos/ShiftDrawer'
    import { SyncStatusBar } from '@/components/pos/SyncStatusBar'
    import { SyncIssuesPanel } from '@/components/pos/SyncIssuesPanel'
    import { computeCartTotals, useCartStore } from '@/lib/store/cart.store'
    import type { ReceiptData } from '@/lib/receipt/encoder'

    export default function PosPage() {
      const { activeShift } = useShiftStore()
      const { items, transactionDiscount } = useCartStore()
      const { total } = computeCartTotals(items, transactionDiscount)

      const [showShiftDrawer, setShowShiftDrawer] = useState(false)
      const [showPayment, setShowPayment] = useState(false)
      const [receiptData, setReceiptData] = useState<ReceiptData | null>(null)
      const [pendingCount, setPendingCount] = useState(0)
      const [isSyncing, setIsSyncing] = useState(false)

      // Register sync listener
      useEffect(() => {
        const cleanup = startSyncListener((results) => {
          setIsSyncing(false)
          // re-count pending from DB (useLiveQuery in SyncStatusBar handles this reactively)
        })
        return cleanup
      }, [])

      // Gate: show shift drawer if no active shift
      if (!activeShift) {
        return (
          <div className="flex h-screen items-center justify-center bg-gray-50">
            <div className="text-center">
              <h1 className="text-2xl font-bold mb-4">Buka Shift untuk Mulai</h1>
              <button
                onClick={() => setShowShiftDrawer(true)}
                className="px-6 py-3 bg-blue-600 text-white rounded-lg text-lg font-medium hover:bg-blue-700"
              >
                Buka Shift
              </button>
              <ShiftDrawer isOpen={showShiftDrawer} onClose={() => setShowShiftDrawer(false)} />
            </div>
          </div>
        )
      }

      return (
        <div className="flex flex-col h-screen bg-gray-50">
          <SyncStatusBar pendingCount={pendingCount} isSyncing={isSyncing} />
          <SyncIssuesPanel />
          <div className="flex flex-1 overflow-hidden">
            <div className="flex-1 overflow-hidden p-4">
              <ProductPanel />
            </div>
            <div className="w-96 border-l border-gray-200 bg-white flex flex-col">
              <CartPanel
                onPay={() => setShowPayment(true)}
              />
            </div>
          </div>
          {/* Header with shift info + close shift button */}
          <div className="absolute top-0 right-0 p-2">
            <button
              onClick={() => setShowShiftDrawer(true)}
              className="text-sm text-gray-600 hover:text-gray-900 underline"
            >
              Shift: {activeShift.id.slice(0, 8)} — Tutup
            </button>
          </div>
          <ShiftDrawer isOpen={showShiftDrawer} onClose={() => setShowShiftDrawer(false)} />
          <PaymentModal
            isOpen={showPayment}
            total={total}
            onSuccess={(txId, receipt) => {
              setShowPayment(false)
              setReceiptData(receipt)
            }}
            onClose={() => setShowPayment(false)}
          />
          <ReceiptModal
            isOpen={receiptData !== null}
            receiptData={receiptData}
            onClose={() => setReceiptData(null)}
          />
        </div>
      )
    }
    ```

    Note: CartPanel needs an onPay prop added (update CartPanel.tsx BAYAR button to call onPay instead of void placeholder). Update CartPanel's Props type to add onPay: () => void.

    Also: SyncStatusBar needs a live count from Dexie. Update SyncStatusBar to use useLiveQuery internally for the pending count (rather than relying on parent prop). Simplify its Props to just { isSyncing: boolean } and compute pendingCount internally via:
    ```typescript

    ```
    Then remove the pendingCount prop from the page.
