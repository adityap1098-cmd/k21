  <name>Task 1: Void service — voidTransaction()</name>
  <files>
    apps/api/src/modules/pos/void.service.ts
  </files>
  <behavior>
    - voidTransaction({ transactionId, reason, performedBy, ipAddress }): Promise<Transaction>
      1. Open db.transaction()
      2. SELECT transaction FOR UPDATE — throw TRANSACTION_NOT_FOUND if missing
      3. If status is already 'VOIDED' — throw ALREADY_VOIDED
      4. UPDATE transactions SET status = 'VOIDED', voidReason = reason, voidedAt = now() WHERE id = transactionId
      5. For each transaction_item: call recordMovement({ variantId, movementType: 'RETURN', qty: Math.abs(item.qty), reference: transactionId, performedBy }, tx) then UPDATE product_variants SET stock_qty = stock_qty + qty WHERE id = variantId
      6. Call createJournalEntryReversal({ transactionId, total: transaction.total }, tx)
      7. Call logAudit({ userId: performedBy, action: 'VOID_TRANSACTION', resourceType: 'transaction', resourceId: transactionId, oldValue: { status: 'COMPLETED' }, newValue: { status: 'VOIDED', reason }, ipAddress })
      8. Return updated transaction
  </behavior>
