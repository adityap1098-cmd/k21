**apps/api/src/queues/redis.ts** — Two separate ioredis connections (BullMQ requires its own connection with specific options; sharing with cache causes BullMQ to hold the connection in subscriber mode):
```typescript
import Redis from 'ioredis'

// BullMQ connection — requires maxRetriesPerRequest: null and enableReadyCheck: false
export const bullmqRedis = new Redis(process.env.REDIS_URL ?? 'redis://localhost:6379', {
  maxRetriesPerRequest: null,
  enableReadyCheck: false,
})

// Cache connection — general purpose ioredis for stock cache reads/writes
export const cacheRedis = new Redis(process.env.REDIS_URL ?? 'redis://localhost:6379', {
  maxRetriesPerRequest: 3,
  enableReadyCheck: true,
})
```
Note: No fail-fast throw for missing REDIS_URL — the existing pattern in apps/api/src/db/index.ts only throws for DATABASE_URL. Redis falls back to localhost in dev/test.
