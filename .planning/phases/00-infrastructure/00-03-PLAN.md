PgBouncer ini config (pool_mode = transaction is REQUIRED):
```ini
[databases]
k21 = host=postgres port=5432 dbname=k21

[pgbouncer]
listen_addr = 0.0.0.0
listen_port = 5432
auth_type = md5
auth_file = /etc/pgbouncer/userlist.txt
pool_mode = transaction
max_client_conn = 100
default_pool_size = 20
min_pool_size = 5
reserve_pool_size = 5
reserve_pool_timeout = 3
server_idle_timeout = 600
log_connections = 0
log_disconnections = 0
```

MD5 hash format for userlist.txt: prefix "md5" + md5sum of (password + username).
Example: password="secret", username="appuser" → echo -n "secretappuser" | md5sum → prefix "md5".

Log rotation (MUST appear on every service):
```yaml
logging:
  driver: "json-file"
  options:
    max-size: "10m"
    max-file: "5"
```

Network rule: ONLY Nginx gets published ports in prod. PgBouncer and Redis MUST NOT have `ports:` in any Compose file.

Dev override naming: MUST be `docker-compose.dev.yml` (NOT `docker-compose.override.yml`). Docker auto-merges override.yml which would contaminate prod runs.

    Key rules:
    - All services on `k21-net` bridge network
    - Log rotation on EVERY service (max-size: 10m, max-file: 5)
    - `depends_on` with `condition: service_healthy` for startup ordering
    - No `ports:` on postgres, pgbouncer, or redis (internal only)
    - `build.context: .` (monorepo root) for api and web — required for pnpm workspace Docker builds
    - Backup service uses a custom Dockerfile (placeholder until Plan 00-06)

    ```yaml
    # docker-compose.yml — base definitions, always merged with prod or dev override
    # Usage:
    #   Production: docker compose -f docker-compose.yml -f docker-compose.prod.yml up -d
    #   Development: docker compose -f docker-compose.yml -f docker-compose.dev.yml up
    # NEVER run this file alone — it is a base config, not a complete config.

    services:
      postgres:
        image: postgres:16-alpine
        environment:
          POSTGRES_USER: ${POSTGRES_USER}
          POSTGRES_PASSWORD: ${POSTGRES_PASSWORD}
          POSTGRES_DB: ${POSTGRES_DB}
        volumes:
          - postgres_data:/var/lib/postgresql/data
        networks:
          - k21-net
        healthcheck:
          test: ["CMD-SHELL", "pg_isready -U ${POSTGRES_USER} -d ${POSTGRES_DB}"]
          interval: 10s
          timeout: 5s
          retries: 5
          start_period: 30s
        logging:
          driver: "json-file"
          options:
            max-size: "10m"
            max-file: "5"

      pgbouncer:
        image: edoburu/pgbouncer:1.23-p1
        environment:
          DATABASE_URL: postgres://${POSTGRES_USER}:${POSTGRES_PASSWORD}@postgres:5432/${POSTGRES_DB}
        volumes:
          - ./pgbouncer/pgbouncer.ini:/etc/pgbouncer/pgbouncer.ini:ro
          - ./pgbouncer/userlist.txt:/etc/pgbouncer/userlist.txt:ro
        networks:
          - k21-net
        depends_on:
          postgres:
            condition: service_healthy
        healthcheck:
          test: ["CMD-SHELL", "pg_isready -h localhost -p 5432"]
          interval: 10s
          timeout: 5s
          retries: 5
          start_period: 10s
        logging:
          driver: "json-file"
          options:
            max-size: "10m"
            max-file: "5"

      redis:
        image: redis:7-alpine
        command: >
          redis-server
          --requirepass ${REDIS_PASSWORD}
          --maxmemory-policy noeviction
        networks:
          - k21-net
        healthcheck:
          test: ["CMD", "redis-cli", "-a", "${REDIS_PASSWORD}", "ping"]
          interval: 10s
          timeout: 3s
          retries: 5
          start_period: 10s
        logging:
          driver: "json-file"
          options:
            max-size: "10m"
            max-file: "5"

      api:
        build:
          context: .
          dockerfile: apps/api/Dockerfile
        environment:
          DATABASE_URL: ${DATABASE_URL}
          REDIS_HOST: redis
          REDIS_PORT: "6379"
          REDIS_PASSWORD: ${REDIS_PASSWORD}
          PORT: "3001"
        networks:
          - k21-net
        depends_on:
          pgbouncer:
            condition: service_healthy
          redis:
            condition: service_healthy
        healthcheck:
          test: ["CMD-SHELL", "curl -f http://localhost:3001/health || exit 1"]
          interval: 30s
          timeout: 5s
          retries: 3
          start_period: 30s
        logging:
          driver: "json-file"
          options:
            max-size: "10m"
            max-file: "5"

      web:
        build:
          context: .
          dockerfile: apps/web/Dockerfile
        environment:
          NODE_ENV: ${NODE_ENV:-production}
        networks:
          - k21-net
        depends_on:
          api:
            condition: service_healthy
        logging:
          driver: "json-file"
          options:
            max-size: "10m"
            max-file: "5"

      backup:
        build:
          context: ./backup
          dockerfile: Dockerfile
        environment:
          DATABASE_URL: ${DATABASE_URL}
          GPG_PASSPHRASE: ${GPG_PASSPHRASE}
          B2_KEY_ID: ${B2_KEY_ID}
          B2_APPLICATION_KEY: ${B2_APPLICATION_KEY}
          B2_BUCKET: ${B2_BUCKET}
        networks:
          - k21-net
        depends_on:
          pgbouncer:
            condition: service_healthy
        logging:
          driver: "json-file"
          options:
            max-size: "10m"
            max-file: "5"

    volumes:
      postgres_data:

    networks:
      k21-net:
        driver: bridge
    ```

    **docker-compose.prod.yml** — adds Nginx, certbot, Netdata; sets restart policies and NODE_ENV:
    ```yaml
    # docker-compose.prod.yml — production overrides
    # Usage: docker compose -f docker-compose.yml -f docker-compose.prod.yml up -d

    services:
      nginx:
        image: nginx:1.25-alpine
        ports:
          - "80:80"
          - "443:443"
        volumes:
          - ./nginx/nginx.conf:/etc/nginx/nginx.conf:ro
          - ./nginx/certbot/conf:/etc/letsencrypt:ro
          - ./nginx/certbot/www:/var/www/certbot:ro
        networks:
          - k21-net
        depends_on:
          api:
            condition: service_healthy
          web:
            condition: service_started
        restart: unless-stopped
        logging:
          driver: "json-file"
          options:
            max-size: "10m"
            max-file: "5"

      certbot:
        image: certbot/certbot
        volumes:
          - ./nginx/certbot/conf:/etc/letsencrypt
          - ./nginx/certbot/www:/var/www/certbot
        entrypoint: >
          /bin/sh -c
          'trap exit TERM;
          while :; do
            certbot renew --webroot -w /var/www/certbot --quiet;
            sleep 12h & wait $${!};
          done'
        logging:
          driver: "json-file"
          options:
            max-size: "10m"
            max-file: "5"

      netdata:
        image: netdata/netdata
        pid: host
        network_mode: host
        cap_add:
          - SYS_PTRACE
          - SYS_ADMIN
        security_opt:
          - apparmor:unconfined
        volumes:
          - netdataconfig:/etc/netdata
          - netdatalib:/var/lib/netdata
          - netdatacache:/var/cache/netdata
          - /etc/passwd:/host/etc/passwd:ro
          - /etc/group:/host/etc/group:ro
          - /proc:/host/proc:ro
          - /sys:/host/sys:ro
          - /etc/os-release:/host/etc/os-release:ro
          - /var/run/docker.sock:/var/run/docker.sock:ro
        environment:
          - NETDATA_CLAIM_TOKEN=
          - DOCKER_HOST=unix:///var/run/docker.sock
        restart: unless-stopped
        logging:
          driver: "json-file"
          options:
            max-size: "10m"
            max-file: "5"

      postgres:
        restart: unless-stopped

      pgbouncer:
        restart: unless-stopped

      redis:
        restart: unless-stopped

      api:
        restart: unless-stopped
        environment:
          NODE_ENV: production

      web:
        restart: unless-stopped
        environment:
          NODE_ENV: production

      backup:
        restart: unless-stopped

    volumes:
      netdataconfig:
      netdatalib:
      netdatacache:
    ```

    **docker-compose.dev.yml** — hot-reload mounts, direct port access, no Nginx:
    ```yaml
    # docker-compose.dev.yml — development overrides
    # Usage: docker compose -f docker-compose.yml -f docker-compose.dev.yml up
    # IMPORTANT: Named docker-compose.dev.yml (NOT docker-compose.override.yml)
    # so it is NOT auto-applied by Docker — requires explicit -f flag.

    services:
      api:
        volumes:
          - ./apps/api/src:/app/src
        ports:
          - "3001:3001"
        environment:
          NODE_ENV: development

      web:
        volumes:
          - ./apps/web/src:/app/src
        ports:
          - "3000:3000"
        environment:
          NODE_ENV: development

      postgres:
        ports:
          - "127.0.0.1:5433:5432"

      redis:
        ports:
          - "127.0.0.1:6379:6379"
    ```

    Note: In dev, postgres is exposed on host port 5433 (not 5432 to avoid conflict with local postgres installs) and redis on 6379 — both bound to 127.0.0.1 only. This is for developer tooling access only. The API still connects through PgBouncer even in dev.
